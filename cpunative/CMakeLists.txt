cmake_minimum_required(VERSION 3.25)
set(SOLPROP_CPUNATIVE_PROFILE "LOCAL" CACHE STRING "LOCAL or DISTRIBUTION")
if(CMAKE_HOST_APPLE AND SOLPROP_CPUNATIVE_PROFILE STREQUAL "DISTRIBUTION")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version")
endif()
project(solprop_cpunative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SOLPROP_CPUNATIVE_TARGET "" CACHE STRING "Explicit local CPU architecture")
option(SOLPROP_CPUNATIVE_SANITIZE "Address and undefined-behaviour sanitizers" OFF)
option(SOLPROP_CPUNATIVE_VECTOR_REPORT "Emit compiler vectorisation diagnostics" OFF)
set(SOLPROP_CPUNATIVE_MATH_PROFILE "PROTECTED" CACHE STRING
  "Experimental math flags: PROTECTED, ERRNO_FREE, NONTRAPPING or FORTRAN_COMPLEX")
set_property(CACHE SOLPROP_CPUNATIVE_MATH_PROFILE PROPERTY STRINGS
  PROTECTED ERRNO_FREE NONTRAPPING FORTRAN_COMPLEX)
set(SOLPROP_CPUNATIVE_RATE_MATH "LIBM" CACHE STRING
  "Experimental ionisation-rate math: LIBM, SVML, SLEEF or MKL")
find_package(Python 3.14 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
  OUTPUT_VARIABLE nanobind_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(nanobind CONFIG REQUIRED)
# The FFTW OpenMP library and this extension must resolve the same runtime.
if(APPLE AND EXISTS "${FFTW_ROOT}/lib/libomp.dylib" AND NOT OpenMP_omp_LIBRARY)
  set(OpenMP_omp_LIBRARY "${FFTW_ROOT}/lib/libomp.dylib" CACHE FILEPATH "Matched FFTW OpenMP runtime")
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
set(SOLPROP_CPUNATIVE_FFT_PROVIDER "FFTW" CACHE STRING "Experimental FFT provider: FFTW or MKL")
set(SOLPROP_CPUNATIVE_BLAS_PROVIDER "AUTO" CACHE STRING "Experimental BLAS provider: AUTO or MKL")
if(SOLPROP_CPUNATIVE_FFT_PROVIDER STREQUAL "MKL" OR SOLPROP_CPUNATIVE_BLAS_PROVIDER STREQUAL "MKL" OR SOLPROP_CPUNATIVE_RATE_MATH STREQUAL "MKL")
  find_path(CPUNATIVE_MKL_INCLUDE mkl.h HINTS "${MKL_ROOT}" ENV MKLROOT PATH_SUFFIXES include REQUIRED)
  find_library(CPUNATIVE_MKL_LIBRARY mkl_rt HINTS "${MKL_ROOT}" ENV MKLROOT PATH_SUFFIXES lib lib/intel64 REQUIRED)
endif()
if(SOLPROP_CPUNATIVE_FFT_PROVIDER STREQUAL "MKL")
  set(FFTW_INCLUDE_DIR "${CPUNATIVE_MKL_INCLUDE}/fftw")
  set(FFTW_LIBRARY "${CPUNATIVE_MKL_LIBRARY}")
  set(FFTW_THREAD_LIBRARY "")
  set(SOLPROP_CPUNATIVE_FFTW_THREADING "MKL")
elseif(SOLPROP_CPUNATIVE_FFT_PROVIDER STREQUAL "FFTW")
find_path(FFTW_INCLUDE_DIR fftw3.h HINTS "${FFTW_ROOT}" PATH_SUFFIXES include REQUIRED)
find_library(FFTW_LIBRARY NAMES fftw3 HINTS "${FFTW_ROOT}" PATH_SUFFIXES lib REQUIRED)
set(SOLPROP_CPUNATIVE_FFTW_THREADING "OPENMP" CACHE STRING "FFTW runtime: OPENMP or PTHREADS")
if(SOLPROP_CPUNATIVE_FFTW_THREADING STREQUAL "OPENMP")
  set(_fftw_thread_name fftw3_omp)
elseif(SOLPROP_CPUNATIVE_FFTW_THREADING STREQUAL "PTHREADS")
  # A common pthread FFTW build permits a compiler-only experiment without
  # bringing another compiler's OpenMP runtime into the process.
  set(_fftw_thread_name fftw3_threads)
else()
  message(FATAL_ERROR "Unknown FFTW threading runtime")
endif()
find_library(FFTW_${SOLPROP_CPUNATIVE_FFTW_THREADING}_LIBRARY NAMES ${_fftw_thread_name}
  HINTS "${FFTW_ROOT}" PATH_SUFFIXES lib REQUIRED)
set(FFTW_THREAD_LIBRARY "${FFTW_${SOLPROP_CPUNATIVE_FFTW_THREADING}_LIBRARY}")
else()
  message(FATAL_ERROR "Unknown experimental FFT provider")
endif()
set(BLA_SIZEOF_INTEGER 4)
if(SOLPROP_CPUNATIVE_BLAS_PROVIDER STREQUAL "MKL")
  set(BLAS_FOUND TRUE)
else()
if(SOLPROP_CPUNATIVE_BLAS_ROOT)
  list(PREPEND CMAKE_PREFIX_PATH "${SOLPROP_CPUNATIVE_BLAS_ROOT}")
endif()
find_package(BLAS)
if(BLAS_FOUND AND NOT APPLE)
  find_path(CBLAS_INCLUDE_DIR cblas.h HINTS "${SOLPROP_CPUNATIVE_BLAS_ROOT}" PATH_SUFFIXES include include/openblas REQUIRED)
endif()
endif()

nanobind_add_module(_core NB_STATIC NOMINSIZE src/bindings.cpp src/core.cpp src/beam.cpp src/helpers.cpp src/rates.cpp)
if(SOLPROP_CPUNATIVE_RATE_MATH STREQUAL "SVML")
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
    message(FATAL_ERROR "The isolated SVML rate experiment requires IntelLLVM")
  endif()
  # Keep the altered math-library contract confined to rate interpolation. The
  # protected fraction, scan, and integrator helpers retain their own libm calls.
  set_property(SOURCE src/rates.cpp APPEND PROPERTY COMPILE_OPTIONS
    -fno-lto -fimf-use-svml=true:log,exp -fimf-max-error=1.0:log,exp)
elseif(SOLPROP_CPUNATIVE_RATE_MATH STREQUAL "SLEEF")
  find_path(CPUNATIVE_SLEEF_INCLUDE sleef.h HINTS "${SLEEF_ROOT}" PATH_SUFFIXES include REQUIRED)
  find_library(CPUNATIVE_SLEEF_LIBRARY sleef HINTS "${SLEEF_ROOT}" PATH_SUFFIXES lib lib64 REQUIRED)
  target_include_directories(_core PRIVATE "${CPUNATIVE_SLEEF_INCLUDE}")
  target_link_libraries(_core PRIVATE "${CPUNATIVE_SLEEF_LIBRARY}")
  target_compile_definitions(_core PRIVATE CPUNATIVE_RATE_SLEEF=1)
elseif(SOLPROP_CPUNATIVE_RATE_MATH STREQUAL "MKL")
  target_link_libraries(_core PRIVATE "${CPUNATIVE_MKL_LIBRARY}")
  target_compile_definitions(_core PRIVATE CPUNATIVE_RATE_MKL=1)
elseif(NOT SOLPROP_CPUNATIVE_RATE_MATH STREQUAL "LIBM")
  message(FATAL_ERROR "Unknown experimental rate math provider")
endif()
target_include_directories(_core PRIVATE include "${FFTW_INCLUDE_DIR}")
target_link_libraries(_core PRIVATE OpenMP::OpenMP_CXX "${FFTW_THREAD_LIBRARY}" "${FFTW_LIBRARY}" ${CMAKE_DL_LIBS})
if(SOLPROP_CPUNATIVE_FFT_PROVIDER STREQUAL "MKL" OR SOLPROP_CPUNATIVE_BLAS_PROVIDER STREQUAL "MKL" OR SOLPROP_CPUNATIVE_RATE_MATH STREQUAL "MKL")
  target_include_directories(_core PRIVATE "${CPUNATIVE_MKL_INCLUDE}")
endif()
if(SOLPROP_CPUNATIVE_FFT_PROVIDER STREQUAL "MKL")
  target_compile_definitions(_core PRIVATE CPUNATIVE_MKL_FFT=1)
endif()
if(BLAS_FOUND)
  if(SOLPROP_CPUNATIVE_BLAS_PROVIDER STREQUAL "MKL")
    target_link_libraries(_core PRIVATE "${CPUNATIVE_MKL_LIBRARY}")
    target_compile_definitions(_core PRIVATE CPUNATIVE_MKL_BLAS=1)
  else()
  target_link_libraries(_core PRIVATE BLAS::BLAS)
  endif()
  target_compile_definitions(_core PRIVATE CPUNATIVE_BLAS=1)
  if(APPLE)
    target_compile_definitions(_core PRIVATE ACCELERATE_NEW_LAPACK)
  elseif(NOT SOLPROP_CPUNATIVE_BLAS_PROVIDER STREQUAL "MKL")
    target_include_directories(_core PRIVATE "${CBLAS_INCLUDE_DIR}")
  endif()
endif()
target_compile_options(_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  # Intel's default fast model is unsuitable for the compensated defects. Do not
  # let its startup or generated code flush subnormal values during this study.
  target_compile_options(_core PRIVATE -fp-model=precise -no-ftz -Wno-overriding-option)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|IntelLLVM")
  target_compile_options(_core PRIVATE -ffp-contract=fast-honor-pragmas)
  if(SOLPROP_CPUNATIVE_VECTOR_REPORT)
    target_compile_options(_core PRIVATE -Rpass=loop-vectorize -Rpass-missed=loop-vectorize)
  endif()
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14)
    message(FATAL_ERROR "cpunative requires GCC 14 or newer")
  endif()
  target_compile_options(_core PRIVATE -ffp-contract=fast)
  if(SOLPROP_CPUNATIVE_VECTOR_REPORT)
    target_compile_options(_core PRIVATE -fopt-info-vec-optimized -fopt-info-vec-missed)
  endif()
endif()
# Keep the ablations after -fno-fast-math, which otherwise resets some of them.
# None enables reassociation, reciprocal approximations, finite-only arithmetic,
# or contraction in the protected helpers. The default remains the qualified path.
set(_math_flags "")
if(SOLPROP_CPUNATIVE_MATH_PROFILE MATCHES "^(ERRNO_FREE|NONTRAPPING|FORTRAN_COMPLEX)$")
  list(APPEND _math_flags -fno-math-errno)
  if(NOT SOLPROP_CPUNATIVE_MATH_PROFILE STREQUAL "ERRNO_FREE")
    list(APPEND _math_flags -fno-trapping-math)
  endif()
  if(SOLPROP_CPUNATIVE_MATH_PROFILE STREQUAL "FORTRAN_COMPLEX")
    list(APPEND _math_flags -fcx-fortran-rules)
  endif()
elseif(NOT SOLPROP_CPUNATIVE_MATH_PROFILE STREQUAL "PROTECTED")
  message(FATAL_ERROR "Unknown cpunative math profile: ${SOLPROP_CPUNATIVE_MATH_PROFILE}")
endif()
include(CheckCXXCompilerFlag)
foreach(_flag IN LISTS _math_flags)
  string(MAKE_C_IDENTIFIER "CPUNATIVE_HAS_${_flag}" _flag_test)
  check_cxx_compiler_flag("${_flag}" "${_flag_test}")
  if(NOT ${_flag_test})
    message(FATAL_ERROR "${CMAKE_CXX_COMPILER_ID} does not support ${_flag}")
  endif()
endforeach()
target_compile_options(_core PRIVATE ${_math_flags})
if(SOLPROP_CPUNATIVE_PROFILE STREQUAL "LOCAL")
  if(SOLPROP_CPUNATIVE_TARGET)
    set(_target "${SOLPROP_CPUNATIVE_TARGET}")
  else()
    set(_target native)
  endif()
elseif(SOLPROP_CPUNATIVE_PROFILE STREQUAL "DISTRIBUTION")
  if(APPLE)
    set(_target apple-m1)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(_target neoverse-n1)
  else()
    set(_target x86-64-v3)
  endif()
else()
  message(FATAL_ERROR "Unknown cpunative build profile")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  target_compile_options(_core PRIVATE "-mcpu=${_target}")
else()
  target_compile_options(_core PRIVATE "-march=${_target}")
endif()
if(SOLPROP_CPUNATIVE_SANITIZE)
  target_compile_options(_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(_core PRIVATE -fsanitize=address,undefined)
else()
  include(CheckIPOSupported)
  check_ipo_supported(RESULT _ipo OUTPUT _ipo_error)
  if(_ipo)
    set_property(TARGET _core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  endif()
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
  OUTPUT_VARIABLE _commit OUTPUT_STRIP_TRAILING_WHITESPACE)
file(GLOB_RECURSE _sources CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/include/*.hpp" "${PROJECT_SOURCE_DIR}/src/*.hpp"
  "${PROJECT_SOURCE_DIR}/src/*.cpp")
set(_hash_input "")
foreach(_source IN LISTS _sources)
  file(SHA256 "${_source}" _hash)
  string(APPEND _hash_input "${_hash}")
endforeach()
string(SHA256 _source_hash "${_hash_input}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_sources})
target_compile_definitions(_core PRIVATE
  CPUNATIVE_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}"
  CPUNATIVE_MATH_PROFILE="${SOLPROP_CPUNATIVE_MATH_PROFILE}"
  CPUNATIVE_RATE_MATH="${SOLPROP_CPUNATIVE_RATE_MATH}"
  CPUNATIVE_FFTW_THREADING="${SOLPROP_CPUNATIVE_FFTW_THREADING}"
  CPUNATIVE_PROFILE="${SOLPROP_CPUNATIVE_PROFILE}" CPUNATIVE_TARGET="${_target}"
  CPUNATIVE_COMMIT="${_commit}" CPUNATIVE_SOURCE_HASH="${_source_hash}"
  CPUNATIVE_IPO=$<AND:$<CONFIG:Release>,$<BOOL:${_ipo}>>)
# Archive this alongside measurements: it includes the effective target options,
# while compile_commands.json also records CMake's compiler-specific IPO flags.
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/compile-options-$<CONFIG>.txt"
  CONTENT "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\n${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE}\n$<JOIN:$<TARGET_PROPERTY:_core,COMPILE_OPTIONS>, >\n")
install(TARGETS _core LIBRARY DESTINATION solprop_cpunative)
